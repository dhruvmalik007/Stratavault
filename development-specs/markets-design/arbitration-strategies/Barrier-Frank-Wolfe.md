#### **B. The Integer Programming Oracle (Step 2b)**

This is the "expensive" part. It takes the current gradient and finds the vertex that minimizes the correlation.

```typescript
// src/lib/quant/oracle.ts
import GLPK from 'glpk.js';
import { Vector, LinearConstraint, OracleResult } from './types';

export class IPOracle {
  private glpk: any;
  
  constructor() {
    // Initialize GLPK (async in browser/node)
    this.glpk = GLPK(); 
  }

  /**
   * Solves: argmin_{z in Z} (gradient * z)
   * This finds the "descent vertex" - the valid outcome that reduces our cost most.
   */
  public async findDescentVertex(
    gradient: Vector, 
    constraints: LinearConstraint[],
    dimension: number
  ): Promise<OracleResult> {
    const glpk = await this.glpk;

    // 1. Setup the Linear Problem
    const lp = {
      name: 'FrankWolfe_Step',
      objective: {
        direction: glpk.GLP_MIN, // We want to minimize gradient * z
        name: 'obj',
        vars: gradient.map((val, idx) => ({ 
          name: `x_${idx}`, 
          coef: val 
        }))
      },
      subjectTo: constraints.map((c, i) => ({
        name: `cons_${i}`,
        vars: c.coeffs.map((val, idx) => ({ name: `x_${idx}`, coef: val })),
        bnds: this.mapConstraintType(glpk, c.type, c.bound)
      })),
      binaries: gradient.map((_, i) => `x_${i}`) // FORCE Integer Constraints (0 or 1)
    };

    // 2. Solve (The "Hard" Part)
    // In a real tournament, this takes <1s early on, >5s later.
    const result = glpk.solve(lp);

    // 3. Parse Result
    const vertex: Vector = new Array(dimension).fill(0);
    for (let i = 0; i < dimension; i++) {
      vertex[i] = result.result.vars[`x_${i}`] || 0;
    }

    return {
      vertex,
      score: result.result.z
    };
  }

  private mapConstraintType(glpk: any, type: string, val: number) {
    if (type === '<=') return { type: glpk.GLP_UP, ub: val, lb: -Infinity };
    if (type === '>=') return { type: glpk.GLP_LO, ub: Infinity, lb: val };
    return { type: glpk.GLP_FX, ub: val, lb: val };
  }
}

```

#### **C. The Barrier Frank-Wolfe Solver (The Core Logic)**

```typescript
// src/lib/quant/frank-wolf.ts
import { Vector, FWConfig, LinearConstraint } from './types';
import { IPOracle } from './oracle';

export class BarrierFrankWolfe {
  private oracle: IPOracle;
  private activeSet: Vector[] = []; // Z_t: The set of discovered vertices
  private weights: number[] = [];   // Alpha: Weights for the active set
  
  constructor() {
    this.oracle = new IPOracle();
  }

  /**
   * Main Execution Pipeline
   * @param marketPrices The observed market prices (pi)
   * @param constraints The logical rules from the AI Agent
   */
  public async project(
    marketPrices: Vector, 
    constraints: LinearConstraint[], 
    config: FWConfig
  ) {
    const dim = marketPrices.length;
    let epsilon = config.initialEpsilon;
    
    // 1. Initialization: Start with a trivial valid vertex (e.g., all zeros if valid)
    // In practice, we run the Oracle once with a random gradient to get z_0
    const z0 = await this.oracle.findDescentVertex(
        marketPrices.map(() => Math.random() - 0.5), 
        constraints, 
        dim
    );
    this.activeSet = [z0.vertex];
    this.weights = [1.0];

    // The Uniform Distribution (Barrier Center)
    const u: Vector = new Array(dim).fill(0.5); 

    for (let t = 0; t < config.maxIterations; t++) {
      // A. Compute Current Point mu_t (Barycentric combination of active set)
      // mu = Sum(alpha_i * z_i)
      const mu_raw = this.computeCurrentMu(dim);

      // B. Apply Barrier Contraction
      // x_t = (1 - eps) * mu + eps * u
      const x_t = mu_raw.map((val, i) => (1 - epsilon) * val + epsilon * u[i]);

      // C. Compute Gradient at Contracted Point
      // Grad = ln(x_t) - ln(market_prices)
      const gradient = x_t.map((val, i) => Math.log(val) - Math.log(marketPrices[i]));

      // D. The Oracle Step (Solve IP)
      // Find z_t that minimizes <Gradient, z>
      const { vertex: z_t, score: linearScore } = await this.oracle.findDescentVertex(
          gradient, 
          constraints, 
          dim
      );

      // E. Compute Convergence Gap
      // gap = <Gradient, x_t - z_t>
      // Note: x_t is current pos, z_t is where we want to go.
      const dotProduct = gradient.reduce((sum, g, i) => sum + g * x_t[i], 0);
      const gap = dotProduct - linearScore;

      // Log progress
      console.log(`Iter ${t}: Eps=${epsilon.toFixed(4)} Gap=${gap.toFixed(4)} ActiveSet=${this.activeSet.length}`);

      if (gap <= config.tolerance) {
        console.log("Converged via Gap Check");
        break;
      }

      // F. Adaptive Epsilon Update (The "Controlled Growth" logic)
      // Check if we can shrink the barrier safely
      // This implementation uses a simplified heuristic from the paper
      if (gap < epsilon * 10) { // Heuristic check
         epsilon = Math.max(1e-6, epsilon / 2);
      }

      // G. Update Active Set & Weights
      this.updateActiveSet(z_t, gradient);
    }

    return this.computeCurrentMu(dim);
  }

  // Helper: Current position in the polytope
  private computeCurrentMu(dim: number): Vector {
    const mu = new Array(dim).fill(0);
    this.activeSet.forEach((z, i) => {
      const w = this.weights[i];
      for (let j = 0; j < dim; j++) mu[j] += w * z[j];
    });
    return mu;
  }

  /**
   * Step 2a: Solve convex optimization over conv(Z_{t-1})
   * In standard FW, we just do a line search.
   * For fully corrective FW, we re-optimize weights over all active vertices.
   * * SIMPLIFIED: Line Search Method (alpha = 2 / (t + 2))
   * PRO: Very fast. CON: Slower convergence than fully corrective.
   */
  private updateActiveSet(newVertex: Vector, gradient: Vector) {
    // Check if vertex already exists
    const exists = this.activeSet.find(v => this.vectorsEqual(v, newVertex));
    if (!exists) {
      this.activeSet.push(newVertex);
      this.weights.push(0); // Initialize with 0 weight
    }

    // Perform Line Search (or Fully Corrective optimization)
    // Here we use the standard FW step size for clarity: gamma = 2/(k+2)
    // A production version would run a coordinate descent here.
    const k = this.activeSet.length;
    const gamma = 2.0 / (k + 2.0);

    // Decay old weights
    this.weights = this.weights.map(w => w * (1 - gamma));
    // Add weight to new vertex (or the existing one we just found again)
    const idx = exists ? this.activeSet.indexOf(exists) : this.activeSet.length - 1;
    this.weights[idx] += gamma;
  }

  private vectorsEqual(a: Vector, b: Vector): boolean {
    return a.every((val, i) => Math.abs(val - b[i]) < 1e-9);
  }
}
```
